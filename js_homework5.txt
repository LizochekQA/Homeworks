var services = {
  "стрижка": "60 грн",
  "гоління": "80 грн",
  "Миття голови": "100 грн"
};

services["Розбите скло"] = "200 грн";

services.price = function () {
  let sum = 0;

  for (let key in this) {
    const price = parseInt(this[key]);

    if (!isNaN(price)) {
      sum = sum + price;
    }
  }

  return sum;
};
services.minPrice = function () {
  let min;

  for (let key in this) {
    const price = parseInt(this[key]);

    if (!isNaN(price)) {
      if (min === undefined || price < min) {
        min = price;
      }
    }
  }

  return min;
};

services.maxPrice = function () {
  let max;

  for (let key in this) {
    const price = parseInt(this[key]);

    if (!isNaN(price)) {
      if (max === undefined || price > max) {
        max = price;
      }
    }
  }

  return max;
};
console.log( services.price(), services.minPrice(), services.maxPrice())
