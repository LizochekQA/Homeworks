'number' + 3 + 3 // Рядок додається до чисел. Результат: "number33"

null + 3 // null приводиться до числа 0. 0 + 3. Результат: 3

5 && "qwerty" // && повертає перше falsy значення або останнє truthy, якщо всі істинні, у нас значення всі вірні, тому повертає останнє вірне. Результат: "qwerty"

+'40' + +'2' + "hillel" // Приводимо рядки до чисел(бо стоїть + перед рядком з числом) 40 + 2 = 42, далі конкатенація з рядком. Результат: "42hillel"

'10' - 5 === 6; // Рядок приводиться до числа: 10 - 5 = 5. 5 === 6. Результат: false

true + false // true = 1, false = 0. Результат: 1

'4px' - 3 // Рядок, що поєднує числа та літери не може бути приведеним до числа. Результат: NaN

'4' - 3 // Рядок приводимо до числа. 4 - 3. Результат: 1

'6' + 3 ** 0; // Оператор ** має вищий пріоритет. Будь-яке число в нульовому степені дорівнює 1. Далі відбувається конкатенація рядка з числом "6" + 1. Результат: "61"

12 / '6' // Рядок приводиться до числа. 12 / 6. Результат: 2

'10' + (5 === 6); // (5 === 6) це false(невірне значення). Конкатенація рядка з false. '10' +  false. Результат: "10false"

null == '' // При нестрогому порівнянні null дорівнює тільки null. Пустий рядок до null не прирівнюється. Результат: false

3 ** (9 / 3); // Спочатку дія в дужках, а після підведення до степеня. Результат: 27

!!'false' == !!'true' // !! - приводить будь-яке значення до типу boolean. Будь-який не порожній рядок це true, а у нас два не порожніх рядка. Результат: true

0 || '0' && 1 // && повертає перше falsy значення або останнє truthy, якщо всі істинні( '0' && 1 ці два значення true) . || повертає перше вірне значення. Результат: true.

(+null == false) < 1; // +null це 0, false це 0, отже 0==0 це true, true це 1. 1 < 1. Результат: false

false && true || true // &&- пріоритет. && повертає перше хибне значення. || повертає перше вірне значення. Результат: true

false && (false || true); // && повертає перше хибне значення. || повертає перше вірне значення. Спочатку розкриваємо дужки. Результат: false

(+null == false) < 1 ** 5; // (0 == 0) це true(true це 1), 1 в 5 степені це 1. 1 < 1. Результат: false
